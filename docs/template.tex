% -------------------------------------------------------------- title page ---
% metropolis sizes its title-page minipage to \paperheight, which is taller than
% the text area of even a plain frame, so \maketitle overfills its vbox by
% 15.64pt on every deck. Same template, sized to \textheight. The content is
% \vfill-centred either way, so this moves the block down by half the overflow.
\makeatletter
\setbeamertemplate{title page}{
  \begin{minipage}[b][\textheight]{\textwidth}
    \ifx\inserttitlegraphic\@empty\else\usebeamertemplate*{title graphic}\fi
    \vfill%
    \ifx\inserttitle\@empty\else\usebeamertemplate*{title}\fi
    \ifx\insertsubtitle\@empty\else\usebeamertemplate*{subtitle}\fi
    \usebeamertemplate*{title separator}
    \ifx\beamer@shortauthor\@empty\else\usebeamertemplate*{author}\fi
    \ifx\insertdate\@empty\else\usebeamertemplate*{date}\fi
    \ifx\insertinstitute\@empty\else\usebeamertemplate*{institute}\fi
    \vfill
    \vspace*{1mm}
  \end{minipage}
}
\makeatother

% -------------------------------------------------------------- pdfpc ---
% Presenter notes for pdfpc. The notes ride inside this PDF as zero-size PDF
% annotations, so there is still one file and it opens normally in any viewer;
% only pdfpc reads them. No separate .pdfpc sidecar and no double-width
% notes-on-second-screen page geometry.
%
%   overridenote  maps beamer's \note{...} onto \pdfpcnote, so notes are
%                 written with the command beamer users already know. A note
%                 must sit INSIDE its \begin{frame}...\end{frame}; beamer's
%                 after-the-frame \note form does not reach pdfpc.
%   duration      talk length in minutes, drives the presenter-screen timer.
%   lastminutes   timer turns red with this many minutes left.
%
% The note body is written into a PDF string, not typeset, so it takes plain
% text, not LaTeX: no \textbf, no math, no macros. pdfpc renders it as markdown
% (discount), so *emphasis*, **bold** and "- " bullets work.
%
% Line breaks are the one trap. A line break in the source is just a space by
% the time it reaches the PDF string, and markdown joins single newlines into
% one paragraph anyway. Only \\ emits a newline, so every bullet has to end
% with \\ or the whole note renders as one run-on paragraph. Two \\ start a new
% paragraph.
%
% Parentheses are fine in pairs but not singly, and % and # are still
% TeX-special, so write them \% and \#.
\usepackage[overridenote,duration=10,lastminutes=2]{pdfpc}

% =============================================================== demo frame ===
% How a frame works, and how its presenter note works. Delete this frame once
% the deck has real content; keep it while more than one person is editing.
\begin{frame}[fragile]{How a frame is written}
  The argument to \file{frame} is the frame title. Body text is plain LaTeX: a
  blank line starts a new paragraph, and metropolis spaces them for you.

  \begin{itemize}
    \item A bullet list is the usual \file{itemize}
    \item \alert{alert} colours a phrase in the theme's accent
  \end{itemize}

  A fixed-width block, for a command or a config line:

  \begin{lstlisting}
docker run --rm -it alpine sh
  \end{lstlisting}

  \note{
    - This text is a presenter note, shown only by pdfpc on the second screen\\
    - It sits INSIDE the frame environment, just before the frame ends\\
    - Every line ends with a double backslash, or the note renders as one paragraph\\
    - It is plain text, not LaTeX: no macros, no bold command; markdown *emphasis* works\\
    - Escape \% and \# ; keep parentheses in pairs
  }
\end{frame}

\begin{frame}{Frame options worth knowing}
  \begin{itemize}
    \item \file{\textbackslash begin\{frame\}[fragile]} is required when the body
          holds a \file{verbatim} or \file{lstlisting} environment that beamer
          cannot read ahead over
    \item \file{[plain]} drops the frame title, footer and progress bar, for a
          full-bleed figure
    \item \file{[standout]} inverts the frame, for a section break or a closing slide
    \item \file{[allowframebreaks]} splits an overlong frame across several slides
  \end{itemize}

  A figure scaled to whatever space is left:

  \begin{center}
    \begin{adjustbox}{max width=\textwidth, max totalheight=0.35\textheight}
      \begin{tikzpicture}[node distance=12mm]
        \node[box,fill=fill1] (a) {Input};
        \node[box,fill=fill3,right=of a] (b) {Process};
        \node[box,fill=fill2,right=of b] (c) {Output};
        \draw[flow] (a) -- (b);
        \draw[flow] (b) -- (c);
      \end{tikzpicture}
    \end{adjustbox}
  \end{center}

  \note{
    - Second demo frame: frame options and a scaled figure\\
    - adjustbox keeps a figure inside the frame whatever its natural size
  }
\end{frame}
