% -------------------------------------------------------------- pdfpc ---
% Presenter notes for pdfpc. The notes ride inside this PDF as zero-size PDF
% annotations, so there is still one file and it opens normally in any viewer;
% only pdfpc reads them. No separate .pdfpc sidecar and no double-width
% notes-on-second-screen page geometry.
%
%   overridenote  maps beamer's \note{...} onto \pdfpcnote, so notes are
%                 written with the command beamer users already know. A note
%                 must sit INSIDE its \begin{frame}...\end{frame}; beamer's
%                 after-the-frame \note form does not reach pdfpc.
%   duration      talk length in minutes, drives the presenter-screen timer.
%   lastminutes   timer turns red with this many minutes left.
%
% The note body is written into a PDF string, not typeset, so it takes plain
% text, not LaTeX: no \textbf, no math, no macros. pdfpc renders it as markdown
% (discount), so *emphasis*, **bold** and "- " bullets work.
%
% Line breaks are the one trap. A line break in the source is just a space by
% the time it reaches the PDF string, and markdown joins single newlines into
% one paragraph anyway. Only \\ emits a newline, so every bullet has to end
% with \\ or the whole note renders as one run-on paragraph. Two \\ start a new
% paragraph.
%
% Parentheses are fine in pairs but not singly, and % and # are still
% TeX-special, so write them \% and \#.
\usepackage[overridenote,duration=10,lastminutes=2]{pdfpc}

\begin{frame}[fragile]{How a frame is written}
  The argument to \file{frame} is the frame title. Body text is plain LaTeX: a
  blank line starts a new paragraph, and metropolis spaces them for you.

  \begin{itemize}
    \item A bullet list is the usual \file{itemize}
    \item \alert{alert} colours a phrase in the theme's accent
  \end{itemize}

  A fixed-width block, for a command or a config line:

  \begin{lstlisting}
docker run --rm -it alpine sh
  \end{lstlisting}

  \note{
    - This text is a presenter note, shown only by pdfpc on the second screen\\
    - It sits INSIDE the frame environment, just before the frame ends\\
    - Every line ends with a double backslash, or the note renders as one paragraph\\
    - It is plain text, not LaTeX: no macros, no bold command; markdown *emphasis* works\\
    - Escape \% and \# ; keep parentheses in pairs
  }
\end{frame}

\begin{frame}{Frame options worth knowing}
  \begin{itemize}
    \item \file{\textbackslash begin\{frame\}[fragile]} is required when the body
          holds a \file{verbatim} or \file{lstlisting} environment that beamer
          cannot read ahead over
    \item \file{[plain]} drops the frame title, footer and progress bar, for a
          full-bleed figure
    \item \file{[standout]} inverts the frame, for a section break or a closing slide
    \item \file{[allowframebreaks]} splits an overlong frame across several slides
  \end{itemize}

  A figure scaled to whatever space is left:

  \begin{center}
    \begin{adjustbox}{max width=\textwidth, max totalheight=0.35\textheight}
      \begin{tikzpicture}[node distance=12mm]
        \node[box,fill=fill1] (a) {Input};
        \node[box,fill=fill3,right=of a] (b) {Process};
        \node[box,fill=fill2,right=of b] (c) {Output};
        \draw[flow] (a) -- (b);
        \draw[flow] (b) -- (c);
      \end{tikzpicture}
    \end{adjustbox}
  \end{center}

  \note{
    - Second demo frame: frame options and a scaled figure\\
    - adjustbox keeps a figure inside the frame whatever its natural size
  }
\end{frame}

\section{Architecture and device integration}
 
\begin{frame}{System overview}
  EEG in, a focus state out, one policy applied consistently downstream.
 
  \begin{center}
    \begin{adjustbox}{max width=\textwidth, max totalheight=0.32\textheight}
      \begin{tikzpicture}[node distance=8mm and 9mm]
        \node[box,fill=fill1] (muse) {Muse};
        \node[box,fill=fill2,right=of muse] (sig) {Signal\\\footnotesize processing};
        \node[box,fill=fill3,right=of sig] (sm) {State\\\footnotesize machine};
        \node[box,fill=fill4,right=of sm] (dnd) {DND\\\footnotesize suppression};
        \node[box,fill=fill1,right=of dnd] (notif) {Catch-up\\\footnotesize summary};
 
        \draw[flow] (muse) -- (sig);
        \draw[flow] (sig) -- (sm);
        \draw[flow] (sm) -- (dnd);
        \draw[flow] (dnd) -- (notif);
 
        \node[box,fill=fill2,below=10mm of sm] (cal) {Calibration};
        \draw[flow] (cal) -- node[lbl,right]{thresholds} (sm);
      \end{tikzpicture}
    \end{adjustbox}
  \end{center}
 
  \small
  \begin{itemize}
    \item Runs entirely on-device, with no data leaves the phone
    \item One state drives both suppression and the catch up summary
    \item Fail open at every stage, a bad signal never silently holds a notification
  \end{itemize}
 
  \note{
    - EEG in, one focus state out, everything downstream reacts to that single signal consistently\\
    - Walk the pipeline left to right: BLE from the headset -> signal processing (artifact rejection + rolling normalisation) -> focus index -> state machine -> FOCUSED / NOT_FOCUSED\\
    - That state is what both DND suppression and the catch-up summary key off\\
    - Calibration is NOT part of the live loop -- one-time per-user step, shown underneath, fits the thresholds the state machine uses\\
    - Thresholds stored as z-scores not raw numbers, because raw EEG power drifts with electrode placement, hair, even caffeine\\
    - Three closing points: runs entirely on-device at ~1\,Hz, no data leaves the phone; one state drives both suppression and the summary, no separate logic path; fail-open runs through every stage -- bad or missing signal always means "let notifications through", never "hold them"
  }
\end{frame}
 
\begin{frame}{Connecting to the headset}
  A standalone BLE peripheral, not a paired Wear OS watch, so we use Muse's
SDK (\file{LibMuse}) rather than the Wear OS Data Layer API.
 
  \begin{center}
    \begin{adjustbox}{max width=\textwidth, max totalheight=0.28\textheight}
      \begin{tikzpicture}[node distance=8mm and 10mm]
        \node[box,fill=fill1] (muse) {Muse 2};
        \node[box,fill=fill2,right=of muse] (sdk) {LibMuse SDK};
        \node[box,fill=fill3,right=of sdk] (impl) {Adapter};
        \node[box,fill=fill4,right=of impl] (pipe) {Signal pipeline};
 
        \draw[flow] (muse) -- node[lbl,above]{BLE} (sdk);
        \draw[flow] (sdk) -- node[lbl,above]{callbacks} (impl);
        \draw[flow] (impl) -- (pipe);
      \end{tikzpicture}
    \end{adjustbox}
  \end{center}
 
  \small
  \begin{itemize}
    \item Registers only the packet types needed, nothing Muse specific leaks out
    \item Small connection state machine, rescans automatically on a drop
    \item A dropped or poor contact signal reads as unknown, never silently nothing
  \end{itemize}
 
  \note{
    - Specifically how we talk to the device, following on from the pipeline overview\\
    - Muse is a standalone BLE peripheral, not a paired Wear OS watch -- that's why vendor SDK (LibMuse) over the Data Layer API\\
    - Our adapter registers only the specific packet types the pipeline needs -- nothing Muse-specific leaks out past that boundary, everything above just sees our own signal pipeline\\
    - Small connection state machine underneath handles rescanning automatically on a BLE drop\\
    - Ties back to fail-open from slide one -- a dropped or poor-contact signal reads as unknown, same as any other bad-signal case, never silently nothing
  }
\end{frame}
 
\begin{frame}[fragile]{Built as four independent modules}
  The pipeline's own stages became the team's module boundaries — one
  interface per stage, one owner, developed and tested in isolation.
 
  \begin{center}
    \begin{adjustbox}{max width=\textwidth, max totalheight=0.3\textheight}
      \begin{tikzpicture}[node distance=8mm and 9mm]
        \node[box,fill=fill1] (eeg) {Sensing};
        \node[box,fill=fill2,right=of eeg] (engine) {Signal \&\\\footnotesize calibration};
        \node[box,fill=fill3,right=of engine] (supp) {Notifications};
        \node[box,fill=fill4,right=of supp] (ui) {UI \& stats};
 
        \draw[flow] (eeg) -- (engine);
        \draw[flow] (engine) -- (supp);
        \draw[flow] (supp) -- (ui);
        \draw[back] (engine) -- (ui);
        \draw[back] (supp) -- (ui);
      \end{tikzpicture}
    \end{adjustbox}
  \end{center}
 
  \small
  \begin{itemize}
    \item Every interface has access to a fake demo, so all four people started building day one
    \item An OSC bridge to an existing app gave real EEG data before BLE worked
    \item Swapping any module is a one line change, nothing downstream changes
  \end{itemize}
  \note{
    - One more thing worth mentioning since we're a four person team, the pipeline just shown is also how we split the work\\
    - Sensing, signal and calibration, notifications, UI and stats, four pipeline stages, four independent modules, one person per stage\\
    - What made it work on a fixed timeline, every interface had a fake from day one, so nobody blocked on anybody else\\
    - Went further on hardware specifically, bridged an existing EEG streaming app over OSC early on, real EEG data to build and test signal processing against BEFORE our own Bluetooth integration worked\\
    - Swapping any module is a one line change, nothing downstream changes
  }
\end{frame}

\section{Visualisation and summary data}

\begin{frame}{Live view: what the headset is saying right now}
  \begin{columns}[T]
    \begin{column}{0.36\textwidth}
      \begin{center}
        \includegraphics[height=0.74\textheight,keepaspectratio,trim=0 70 0 110,clip]{figures/home.png}
      \end{center}
    \end{column}
    \begin{column}{0.62\textwidth}
      \small
      \begin{itemize}
        \item \alert{Focus index}: the last two minutes at 1\,Hz, in standard
              deviations from the user's own calibration baseline
        \item The shaded band is the hysteresis zone between the two
              thresholds, where nothing changes
        \item Artifact-rejected epochs \emph{break} the line rather than being
              interpolated over; the y-range is fixed at $\pm3$\,SD so a calm
              signal looks calm
        \item State pill with time-in-state, per-electrode contact quality,
              and the held-back queue as ``app $\cdot$ count'', never message
              content
      \end{itemize}
    \end{column}
  \end{columns}

  \note{
    - Home screen is the live view: one glance tells you the state, the signal and whether the headset is actually making contact\\
    - The chart is the focus index for the last 120 seconds, one point per second, in SDs from the user's own calibration baseline, not raw band power\\
    - Shaded band = the hysteresis zone between T\_high and T\_low. Inside it nothing changes, so you can see why the state did or didn't flip\\
    - Two deliberate choices: rejected epochs (blinks, jaw clench) break the line instead of being smoothed over, and the y-axis is a fixed plus or minus 3 SD so a calm signal looks calm instead of auto-scaling into noise\\
    - Below it: contact quality per electrode, and the held-back card shows only app and count. Showing the messages here would be the interruption the app exists to prevent
  }
\end{frame}

\begin{frame}{Summary data: what a week of focus looks like}
  \begin{columns}[T]
    \begin{column}{0.36\textwidth}
      \begin{center}
        \includegraphics[height=0.74\textheight,keepaspectratio,trim=0 70 0 110,clip]{figures/stats.png}
      \end{center}
    \end{column}
    \begin{column}{0.62\textwidth}
      \small
      \begin{itemize}
        \item One record per session: focused time, longest streak,
              notifications held back and delivered
        \item Focused time is measured \alert{between state transitions},
              not by counting FOCUSED epochs, so a dropped packet never
              shortens the total
        \item \alert{When you focus}: focused minutes by hour of day, summed
              across every session, so the user can see when they do their
              best work
        \item Stored on-device as one JSON line per session, written once at
              session end; every total is computed on read, nothing leaves
              the phone
      \end{itemize}
    \end{column}
  \end{columns}

  \note{
    - Stats tab is the summary data. Four numbers at the top: total focus, longest unbroken streak, notifications held back, notifications delivered\\
    - Focused time is measured between transitions into and out of FOCUSED, not by counting focused epochs, so one dropped or rejected packet does not quietly shave seconds off the total\\
    - The chart is focused minutes by hour of day across all sessions. This user clearly does their best work mid-morning, so that is when to put the headset on\\
    - Persistence is deliberately simple: one JSON line per session, appended once when the session ends, kept in the app's private storage. Decisions arrive at 1 Hz for hours, so nothing is written per decision\\
    - All the totals are computed on read from that list. No database, no server, nothing leaves the phone
  }
\end{frame}
